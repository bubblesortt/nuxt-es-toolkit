# v2.1 Setup Performance Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Remove setup-time evaluation of all `es-toolkit` runtime surfaces while preserving the exact v2.0.0 public API and generated imports, with reproducible before/after performance evidence.

**Architecture:** A committed generated catalogue supplies sorted export-name arrays and public method types. The planner consumes those arrays, while the existing runtime barrels remain the Nuxt import sources for pnpm-safe dependency resolution. A subprocess-based benchmark harness compares the base ref and candidate ref on the same Node 24 runner.

**Tech Stack:** TypeScript 5.9, ESM, Nuxt Kit 4.5, Nuxt Module Builder, Vitest 4, Node child processes, GitHub Actions, pnpm 10.

## Global Constraints

- Start from current `main` on branch `feature/v2.1-setup-performance`.
- Keep `es-toolkit` pinned to exactly `1.50.0`.
- Keep Nuxt compatibility at `>=3.0.0`; CI remains Nuxt 3.21.11/Node 20.19.5 and Nuxt 4.5.2/Node 24.14.0.
- Add no public module options and change no generated import names, sources, selection rules, warnings, or errors.
- Keep all five files under `src/runtime/`; application imports must continue to resolve through them.
- Do not add consumer install hooks, runtime package scanning, network access, or a new runtime dependency.
- Use two-space indentation, LF, single quotes, and no semicolons.
- Do not change `package.json` version, create a tag, publish the package, or push a release.
- Release gates: at least 70% lower incremental cold-import median, at least 50% lower incremental RSS, identical normalized imports, exactly 185 default imports for `es-toolkit@1.50.0`, and no more than 1% client/server bundle growth.

Implement these exact exported shapes in `bench/metrics.mjs`:

```js
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { extname, join } from 'node:path'
import { gzipSync } from 'node:zlib'

const round = value => Math.round(value * 1000) / 1000
const compareStrings = (left, right) => left < right ? -1 : left > right ? 1 : 0

export const summarize = (values) => {
  if (values.length === 0) {
    throw new Error('Cannot summarize an empty sample.')
  }
  const sorted = [...values].sort((left, right) => left - right)
  const percentile = ratio => sorted[Math.min(sorted.length - 1, Math.ceil(sorted.length * ratio) - 1)]
  return {
    min: round(sorted[0]),
    max: round(sorted.at(-1)),
    median: round(percentile(0.5)),
    p95: round(percentile(0.95)),
  }
}

export const normalizeToolkitImports = (declaration) => {
  const imports = []
  const linePattern = /export \{([^}]+)\} from ['"][^'"]*\/runtime\/(es-toolkit-[^.'"]+)[^'"]*['"]/g
  for (const match of declaration.matchAll(linePattern)) {
    const source = match[2]
    for (const specifier of match[1].split(',')) {
      const [name, as = name] = specifier.trim().split(/\s+as\s+/)
      imports.push({ name, as, source })
    }
  }
  return imports.sort((left, right) =>
    compareStrings(left.source, right.source)
    || compareStrings(left.as, right.as)
    || compareStrings(left.name, right.name),
  )
}

const listFiles = directory => readdirSync(directory, { withFileTypes: true }).flatMap((entry) => {
  const path = join(directory, entry.name)
  return entry.isDirectory() ? listFiles(path) : [path]
})

export const measureBundleDirectory = (directory, extensions) => {
  const files = listFiles(directory).filter(path => extensions.includes(extname(path)))
  return files.reduce((metric, path) => {
    const contents = readFileSync(path)
    metric.files += 1
    metric.rawBytes += statSync(path).size
    metric.gzipBytes += gzipSync(contents).length
    return metric
  }, { files: 0, rawBytes: 0, gzipBytes: 0 })
}

const reduction = (baseline, candidate) => (baseline - candidate) / baseline
const growth = (baseline, candidate) => (candidate - baseline) / baseline

export const evaluateV21Gates = (baseline, candidate) => [
  {
    name: 'cold-import-reduction',
    passed: reduction(baseline.moduleLoad.incrementalMs.median, candidate.moduleLoad.incrementalMs.median) >= 0.7,
  },
  {
    name: 'rss-reduction',
    passed: reduction(baseline.moduleLoad.incrementalRssMiB.median, candidate.moduleLoad.incrementalRssMiB.median) >= 0.5,
  },
  {
    name: 'import-manifest',
    passed: JSON.stringify(baseline.imports) === JSON.stringify(candidate.imports),
  },
  {
    name: 'client-bundle',
    passed: growth(baseline.bundles.client.rawBytes, candidate.bundles.client.rawBytes) <= 0.01,
  },
  {
    name: 'server-bundle',
    passed: growth(baseline.bundles.server.rawBytes, candidate.bundles.server.rawBytes) <= 0.01,
  },
]
```

```js
import { spawnSync } from 'node:child_process'
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import { dirname, join, resolve } from 'node:path'
import { pathToFileURL } from 'node:url'

  measureBundleDirectory,

  summarize,

export const parseBenchmarkArgs = (argv) => {
  const values = new Map()
  for (let index = 0; index < argv.length; index += 2) {
    values.set(argv[index], argv[index + 1])
  }
  const root = values.get('--root')
  const label = values.get('--label')
  const output = values.get('--output')
  if (!root || !label || !output) {
    throw new Error('Required arguments: --root, --label, and --output.')
  }
  return {
    root,
    label,
    output,
    runs: Number(values.get('--runs') ?? 20),
    prepareRuns: Number(values.get('--prepare-runs') ?? 5),
  }
}

const run = (command, args, options) => {
  const result = spawnSync(command, args, { encoding: 'utf8', ...options })
  if (result.status !== 0) {
    throw new Error(result.stderr || result.stdout || `${command} exited with ${result.status}`)
  }
  return result.stdout
}
```

```js
const surfaceSpecifiers = {
  base: 'es-toolkit',
  compat: 'es-toolkit/compat',
  fp: 'es-toolkit/fp',
  map: 'es-toolkit/map',
  set: 'es-toolkit/set',
}

const exportNames = module => Object.keys(module).sort()

export const discoverExportCatalogue = async () => {
  const packageJson = await import('es-toolkit/package.json', { with: { type: 'json' } })
  const entries = await Promise.all(Object.entries(surfaceSpecifiers).map(async ([surface, specifier]) => [
    surface,
    exportNames(await import(specifier)),
  ]))
  const surfaces = Object.fromEntries(entries)
  for (const [surface, names] of Object.entries(surfaces)) {
    if (names.length === 0 || new Set(names).size !== names.length) {
      throw new Error(`Invalid ${surface} export surface for es-toolkit ${packageJson.default.version}.`)
    }
  }
  return { version: packageJson.default.version, surfaces }
}
```

```ts
export const normalizeGeneratedToolkitImports = (imports: string) => {
  const matches = imports.matchAll(/export \{([^}]+)\} from ['"][^'"]*\/runtime\/(es-toolkit-[^.'"]+)[^'"]*['"]/g)
  return [...matches].flatMap((match) => {
    const source = match[2]
    return match[1].split(',').map((specifier) => {
      const [name, as = name] = specifier.trim().split(/\s+as\s+/)
      return { name, as, source }
    })
  }).sort((left, right) => left.as.localeCompare(right.as))
}
```
